# ModelForm automáticamente maneja FileField
def upload_file(request):
    if request.method == "POST":
        form = ModelFormWithFileField(request.POST, request.FILES)
        if form.is_valid():
            # El archivo se guarda automáticamente
            form.save()
            return HttpResponseRedirect("/success/url/")
    else:
        form = ModelFormWithFileField()
    return render(request, "upload.html", {"form": form})

# models.py
class Document(models.Model):
    title = models.CharField(max_length=200)
    file = models.FileField(upload_to="documents")
    created_at = models.DateTimeField(auto_now_add=True)
🏗️ Estructura del Modelo:
Campo Tipo Descripción
title CharField Título descriptivo del documento
file FileField Archivo - se guarda en "documents/"
created_at DateTimeField Fecha de creación automática
✅ Ventajas del ModelForm:
  • Automático: No necesitas manejar archivos manualmente
  • Validación: Usa las validaciones del modelo
  • Metadatos: Guarda información adicional automáticamente
  • Organización: Archivos organizados por upload_to
  • Integridad: Transaccional con la base de datos
  • Mantenimiento: Fácil de mantener y extender
