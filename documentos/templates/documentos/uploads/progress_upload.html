class ProgressBarUploadHandler(MemoryFileUploadHandler):
    def __init__(self, request=None):
        super().__init__(request)
        self.progress_id = request.GET.get('progress_id')
    
    def new_file(self, *args, **kwargs):
        super().new_file(*args, **kwargs)
        # Inicializar progreso
        upload_progress[self.progress_id] = {
            'uploaded': 0,
            'total': self.content_length,
            'percent': 0,
            'status': 'uploading'
        }
    
    def receive_data_chunk(self, raw_data, start):
        chunk = super().receive_data_chunk(raw_data, start)
        # Actualizar progreso
        progress = upload_progress[self.progress_id]
        progress['uploaded'] = start + len(raw_data)
        progress['percent'] = (progress['uploaded'] / progress['total']) * 100
        return chunk

⚠️ Importante sobre CSRF:

¿Por qué usar @csrf_exempt y @csrf_protect?

  • Los upload handlers deben modificarse antes de acceder a request.POST
  • CsrfViewMiddleware accede a request.POST automáticamente
  • @csrf_exempt permite modificar handlers sin interferencia del middleware
  • @csrf_protect restaura la protección CSRF en la función de procesamiento
